## استفاده

```php
use Tavaan\Engine\I18n\Translator;
use Tavaan\Engine\I18n\NumberFormatter;

// ساده‌ترین حالت
$t = new Translator(locale: 'fa', fallback: 'en');
$t->addLines(['welcome' => 'به توان خوش آمدید'], 'fa');
$t->get('welcome');                       // 'به توان خوش آمدید'

// با placeholder
$t->addLines(['hello' => 'سلام :name'], 'fa');
$t->get('hello', ['name' => 'علی']);      // 'سلام علی'

// Pluralization
$t->addLines(['items' => '{0}سبد خالی|{1}یک قلم|[2,*]:count قلم'], 'fa');
$t->choice('items', 0);                  // 'سبد خالی'
$t->choice('items', 1);                   // 'یک قلم'
$t->choice('items', 5);                   // '۵ قلم'

// Namespaces (e.g., shop::product.title)
$t->addNamespace('shop', '/path/to/shop/lang');
$t->get('shop::product.title');

// بارگذاری از دایرکتوری
$t->loadDirectory('lang/');

// اعداد فارسی و ارز
$fmt = new NumberFormatter('fa');
$fmt->format(1234567);                   // '۱٬۲۳۴٬۵۶۷'
$fmt->format(1234.5, 2);                  // '۱٬۲۳۴٫۵۰'
$fmt->currency(1000, 'IRR');              // '۱٬۰۰۰ ریال'
$fmt->toPersianDigits('12345');           // '۱۲۳۴۵'
$fmt->fromPersianDigits('۱۲۳۴۵');         // '12345'

// تغییر زبان
$t->setLocale('en');
$t->getLocale();                          // 'en'
```
